    <script>
        // Check if user is logged in
        document.addEventListener('DOMContentLoaded', function() {
            const user = JSON.parse(localStorage.getItem('user'));
            
            if (!user) {
                window.location.href = 'login.html';
                return;
            }
            
            // Display user info
            document.getElementById('userFullName').textContent = user.full_name;
            // Display user info
            document.getElementById('userFullName').textContent = user.full_name;

            // Customize role display
            let displayRole = user.role.toUpperCase();
            if (displayRole === 'SUPER_ADMIN') {
                displayRole = 'ADMIN';
            } else if (displayRole === 'ADMIN') {
                displayRole = 'STAFF';
            }
            document.getElementById('userRole').textContent = displayRole;
            
            // Load systems based on user role
            loadSystems(user.role);
        });
        
        function loadSystems(role) {
            const systemsGrid = document.getElementById('systemsGrid');
            const systems = [
                {
                    id: 'reservation',
                    title: 'Facility Reservation',
                    description: 'Manage facility bookings and reservations',
                    icon: '🏟️',
                    file: 'reservation.html'
                },
                {
                    id: 'profiling',
                    title: 'Resident Profiling',
                    description: 'Manage resident information and records',
                    icon: '👥',
                    file: 'profiling.html'
                },
                {
                    id: 'documents',
                    title: 'Document Processing',
                    description: 'Generate barangay certificates and documents',
                    icon: '📄',
                    file: 'documents.html'
                }
            ];
            
            // Add user management system only for super_admin
            if (role === 'super_admin') {
                systems.push({
                    id: 'users',
                    title: 'User Management',
                    description: 'Manage system users and permissions',
                    icon: '👨‍💼',
                    file: 'user_management.html'
                });
            }
            
            systemsGrid.innerHTML = systems.map(system => `
                <div class="system-card ${system.id}" onclick="navigateTo('${system.file}')">
                    <div class="icon">${system.icon}</div>
                    <h3>${system.title}</h3>
                    <p>${system.description}</p>
                </div>
            `).join('');
        }
        
        function navigateTo(file) {
            window.location.href = file;
        }
        
        function logout() {
            localStorage.removeItem('user');
            window.location.href = 'login.html';
        }
    </script>
