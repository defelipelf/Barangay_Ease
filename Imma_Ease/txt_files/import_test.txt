        function readExcelFile(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                
                reader.onload = (e) => {
                    try {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const firstSheet = workbook.SheetNames[0];
                        const worksheet = workbook.Sheets[firstSheet];
                        const jsonData = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: '' });
                        
                        const headers = jsonData[0] || [];
                        const rows = jsonData.slice(1).filter(row => 
                            row.some(cell => cell && cell.toString().trim() !== '')
                        );
                        
                        resolve({ headers, rows });
                    } catch (error) {
                        reject(error);
                    }
                };
                
                reader.onerror = reject;
                reader.readAsArrayBuffer(file);
            });
        }
        
        function detectColumns(headers) {
            const detected = {};
            const lowerHeaders = headers.map(h => h?.toString().toLowerCase().trim() || '');
            
            for (const [field, possibleNames] of Object.entries(COLUMN_MAPPING)) {
                for (let i = 0; i < lowerHeaders.length; i++) {
                    const header = lowerHeaders[i];
                    if (!header) continue;
                    
                    const isMatch = possibleNames.some(name => 
                        header === name || 
                        header.includes(name) || 
                        name.includes(header)
                    );
                    
                    if (isMatch) {
                        detected[field] = {
                            index: i,
                            header: headers[i]
                        };
                        break;
                    }
                }
            }
            
            return detected;
        }

        function parseFullName(fullNameString) {
            if (!fullNameString || typeof fullNameString !== 'string') {
                return { first_name: '', last_name: '', middle_name: '' };
            }
            
            let name = fullNameString.trim().replace(/\s+/g, ' ');
            
            if (name.includes(',')) {
                const parts = name.split(',').map(p => p.trim());
                const lastName = parts[0];
                const rest = parts[1] || '';
                const nameParts = rest.split(' ').filter(p => p);
                
                return {
                    last_name: lastName,
                    first_name: nameParts[0] || '',
                    middle_name: nameParts.slice(1).join(' ') || ''
                };
            }
            
            const parts = name.split(' ').filter(p => p);
            
            if (parts.length === 1) {
                return { first_name: parts[0], last_name: '', middle_name: '' };
            } else if (parts.length === 2) {
                return { first_name: parts[0], last_name: parts[1], middle_name: '' };
            } else {
                return {
                    first_name: parts[0],
                    last_name: parts[parts.length - 1],
                    middle_name: parts.slice(1, -1).join(' ')
                };
            }
        }

                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(formData)
                    });
                    
                    const result = await response.json();
                    
                    if (result.success) {
                        successCount++;
                        return { success: true };
                    } else {
                        errorCount++;
                        errorMessages.push(`Row ${resident.id}: ${result.error || 'Unknown error'}`);
                        return { success: false, error: result.error };
                    }
                } catch (error) {
                    errorCount++;
                    errorMessages.push(`Row ${resident.id}: ${error.message}`);
                    return { success: false, error: error.message };
                }
            });
            
            Promise.all(importPromises).then(() => {
                closeImportModalFunc();
                
                if (successCount > 0) {
                    showMessage(`✅ Successfully imported ${successCount} residents to database!`, 'success');
                }
                
                if (errorCount > 0) {
                    showMessage(`⚠️ ${errorCount} residents failed to import. ${errorMessages.slice(0, 3).join('; ')}`, 'error');
                    console.error('Import errors:', errorMessages);
                }
                
                loadResidents();
                
                importFile.value = '';
                detectionSection.style.display = 'none';
                previewSection.style.display = 'none';
                nameParsingPreview.style.display = 'none';
                parsedExcelData = null;
                detectedColumns = {};
            });
        }
        
        function mapRowToResident(row, id) {
            const resident = {
                id: id,
                first_name: '',
                middle_name: '',
                last_name: '',
                birth_date: '',
                sex: '',
                civil_status: '',
                contact_number: '',
                house_no: '',
                street: '',
                is_pwd: 0,
                is_senior: 0,
                is_solo_parent: 0,
                is_4ps: 0,
                health_info: null,
                education_info: null,
                voter_info: null
            };
            
            if (detectedColumns.full_name) {
                const fullName = row[detectedColumns.full_name.index]?.toString() || '';
                const parsed = parseFullName(fullName);
                resident.last_name = parsed.last_name;
                resident.first_name = parsed.first_name;
                resident.middle_name = parsed.middle_name;
            }
            
            Object.keys(detectedColumns).forEach(field => {
                const col = detectedColumns[field];
                const value = row[col.index]?.toString().trim() || '';
                
                if (field === 'first_name' && !resident.first_name) resident.first_name = value;
                else if (field === 'last_name' && !resident.last_name) resident.last_name = value;
                else if (field === 'middle_name') resident.middle_name = value;
                else if (field === 'birth_date') resident.birth_date = formatExcelDate(value);
                else if (field === 'sex') {
                    const val = value.toLowerCase();
                    if (val === 'm' || val === 'male') resident.sex = 'Male';
                    else if (val === 'f' || val === 'female') resident.sex = 'Female';
                    else resident.sex = value;
                }
                else if (field === 'civil_status') {
                    const val = value.toLowerCase();
                    const statusMap = { 's': 'Single', 'm': 'Married', 'w': 'Widowed', 'sep': 'Separated' };
                    resident.civil_status = statusMap[val] || value;
                }
                else if (field === 'contact_number') resident.contact_number = value;
                else if (field === 'house_no') resident.house_no = value;
                else if (field === 'street') resident.street = value;
                else if (field === 'is_pwd') resident.is_pwd = (value === '1' || value.toLowerCase() === 'yes' || value.toLowerCase() === 'y') ? 1 : 0;
                else if (field === 'is_senior') resident.is_senior = (value === '1' || value.toLowerCase() === 'yes' || value.toLowerCase() === 'y') ? 1 : 0;
                else if (field === 'is_solo_parent') resident.is_solo_parent = (value === '1' || value.toLowerCase() === 'yes' || value.toLowerCase() === 'y') ? 1 : 0;
                else if (field === 'is_4ps') resident.is_4ps = (value === '1' || value.toLowerCase() === 'yes' || value.toLowerCase() === 'y') ? 1 : 0;
                else if (field === 'blood_type' || field === 'vaccination_status') {
                    if (!resident.health_info) resident.health_info = {};
                    resident.health_info[field] = value;
                }
                else if (field === 'occupation') {
                    if (!resident.education_info) resident.education_info = {};
                    resident.education_info.occupation = value;
                }
                else if (field === 'is_registered_voter' || field === 'precinct_number') {
                    if (!resident.voter_info) resident.voter_info = {};
                    if (field === 'is_registered_voter') {
                        resident.voter_info.is_registered = (value === '1' || value.toLowerCase() === 'yes' || value.toLowerCase() === 'y') ? 1 : 0;
                    } else {
                        resident.voter_info.precinct_number = value;
                    }
                }
            });
            
            return resident;
        }
        
        function formatExcelDate(dateValue) {
            if (!dateValue) return '';
            if (/^\d{4}-\d{2}-\d{2}$/.test(dateValue)) return dateValue;
            
            if (!isNaN(dateValue)) {
                try {
                    const date = new Date((dateValue - 25569) * 86400 * 1000);
                    return date.toISOString().split('T')[0];
                } catch (e) {}
            }
            
            try {
                const date = new Date(dateValue);
                if (!isNaN(date.getTime())) {
                    return date.toISOString().split('T')[0];
                }
            } catch (e) {}
            
            return dateValue;
        }

        function validateMobileNumber(input) {
            const value = input.value;
            const digitsOnly = value.replace(/\D/g, '');
            const errorMessage = input.parentElement.querySelector('.error-message');
            
            if (value && digitsOnly.length < 7) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number must be at least 7 digits';
                    errorMessage.classList.add('show');
                }
                return false;
            } else if (value && digitsOnly.length > 15) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number cannot exceed 15 digits';
                    errorMessage.classList.add('show');
                }
                return false;
            } else if (value && !/^[\d\s+\-()]+$/.test(value)) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number can only contain numbers, spaces, +, -, and ()';
                    errorMessage.classList.add('show');
                }
                return false;
            } else {
                input.classList.remove('error');
                if (errorMessage) {
                    errorMessage.classList.remove('show');
                }
                return true;
            }
        }
        
        function validateForm() {
            let isValid = true;
            
            document.querySelectorAll('.form-control.error').forEach(el => {
                el.classList.remove('error');
            });
            document.querySelectorAll('.error-message.show').forEach(el => {
                el.classList.remove('show');
            });
            
            const requiredFields = ['firstName', 'lastName', 'birthDate', 'sex', 'civilStatus', 'houseNo', 'street'];
            requiredFields.forEach(fieldId => {
                const field = document.getElementById(fieldId);
                const errorMessage = field.parentElement.querySelector('.error-message');
                if (field && !field.value.trim()) {
                    field.classList.add('error');
                    if (errorMessage) {
                        errorMessage.textContent = 'This field is required';
                        errorMessage.classList.add('show');
                    }
                    isValid = false;
                }
            });
            
            const mobileInput = document.getElementById('contactNumber');
            if (mobileInput.value.trim()) {
                if (!validateMobileNumber(mobileInput)) {
                    isValid = false;
                }
            }
            
            const emergencyInput = document.getElementById('emergencyContactNumber');
            if (emergencyInput.value.trim()) {
                const digitsOnly = emergencyInput.value.replace(/\D/g, '');
                if (digitsOnly.length > 0 && digitsOnly.length < 7) {
                    emergencyInput.classList.add('error');
                    const errorMessage = emergencyInput.parentElement.querySelector('.error-message');
                    if (errorMessage) {
                        errorMessage.textContent = 'Emergency contact number must be at least 7 digits';
                        errorMessage.classList.add('show');
                    }
                    isValid = false;
                }
            }
            
            const birthDate = new Date(document.getElementById('birthDate').value);
            if (birthDate > new Date()) {
                const birthDateInput = document.getElementById('birthDate');
                birthDateInput.classList.add('error');
                const errorMessage = birthDateInput.parentElement.querySelector('.error-message');
                if (errorMessage) {
                    errorMessage.textContent = 'Birth date cannot be in the future';
                    errorMessage.classList.add('show');
                }
                isValid = false;
            }
            
            const minDate = new Date();
            minDate.setFullYear(minDate.getFullYear() - 120);
            if (birthDate < minDate) {
                const birthDateInput = document.getElementById('birthDate');
                birthDateInput.classList.add('error');
                const errorMessage = birthDateInput.parentElement.querySelector('.error-message');
                if (errorMessage) {
                    errorMessage.textContent = 'Birth date cannot be more than 120 years ago';
                    errorMessage.classList.add('show');
                }
                isValid = false;
            }
            
            return isValid;
        }

        function sanitizePhoneNumber(phoneNumber) {
            if (!phoneNumber || phoneNumber === 'N/A' || phoneNumber === '') return 'N/A';
            
            if (typeof phoneNumber !== 'string') {
                phoneNumber = String(phoneNumber);
            }
            
            const digitsOnly = phoneNumber.replace(/\D/g, '');
            
            if (digitsOnly.length >= 7 && digitsOnly.length <= 15) {
                if (digitsOnly.length <= 7) {
                    return digitsOnly.replace(/(\d{3})(\d+)/, '$1-$2');
                } else if (digitsOnly.length <= 10) {
                    return digitsOnly.replace(/(\d{3})(\d{3})(\d+)/, '($1) $2-$3');
                } else {
                    return digitsOnly.replace(/(\d{1,3})(\d{3})(\d{3})(\d+)/, '+$1 ($2) $3-$4');
                }
            }
            
            if (phoneNumber.length > 30) {
                return phoneNumber.substring(0, 20) + '...';
            }
            
            return phoneNumber;
        }
        
        function sanitizeAddress(address) {
            if (!address || address === 'N/A' || address === '') return 'N/A';
            
            address = address.replace(/@+/g, '');
            address = address.replace(/\s+/g, ' ').trim();
            
            return address || 'N/A';
        }
        
        function sanitizeText(text) {
            if (!text || text === 'N/A' || text === '') return 'N/A';
            
            text = String(text).replace(/[^\w\s\-\.,#]/g, '');
            text = text.replace(/\s+/g, ' ').trim();
            
            return text || 'N/A';
        }
        
        function truncateText(text, maxLength) {
            if (!text || text === 'N/A') return 'N/A';
            if (text.length <= maxLength) return text;
            return text.substring(0, maxLength) + '...';
        }

        function calculateAge(birthDate) {
            if (!birthDate) return 'N/A';
            
            const today = new Date();
            const birth = new Date(birthDate);
            
            if (isNaN(birth.getTime())) return 'N/A';
            
            let age = today.getFullYear() - birth.getFullYear();
            const monthDiff = today.getMonth() - birth.getMonth();
            
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
                age--;
            }
            
            return age >= 0 ? age : 'N/A';
        }

        async function editResident(id) {
            try {
                const response = await fetch(`${ENDPOINTS.GET_RESIDENT}?id=${id}`);
                const result = await response.json();
                
                if (result.success) {
                    const data = result.data;
                    const resident = data.resident;
                    
                    modalTitle.textContent = `Edit Resident: ${resident.first_name} ${resident.last_name}`;
                    
                    document.getElementById('residentId').value = resident.resident_id;
                    document.getElementById('firstName').value = resident.first_name;
                    document.getElementById('middleName').value = resident.middle_name || '';
                    document.getElementById('lastName').value = resident.last_name;
                    document.getElementById('birthDate').value = resident.birth_date;
                    document.getElementById('sex').value = resident.sex;
                    document.getElementById('civilStatus').value = resident.civil_status;
                    document.getElementById('contactNumber').value = resident.contact_number || '';
                    document.getElementById('houseNo').value = resident.house_no;
                    document.getElementById('street').value = resident.street;
                    document.getElementById('isPwd').checked = resident.is_pwd === 1;
                    document.getElementById('isSenior').checked = resident.is_senior === 1;
                    document.getElementById('isSoloParent').checked = resident.is_solo_parent === 1;
                    document.getElementById('is4ps').checked = resident.is_4ps === 1;
                    
                    if (data.health_info) {
                        document.getElementById('bloodType').value = data.health_info.blood_type || '';
                        document.getElementById('vaccinationStatus').value = data.health_info.vaccination_status || '';
                        document.getElementById('knownAllergies').value = data.health_info.known_allergies || '';
                        document.getElementById('preExistingConditions').value = data.health_info.pre_existing_conditions || '';
                        document.getElementById('emergencyContactName').value = data.health_info.emergency_contact_name || '';
                        document.getElementById('emergencyContactNumber').value = data.health_info.emergency_contact_number || '';
                    }
                    
                    if (data.education_info) {
                        document.getElementById('highestEducation').value = data.education_info.highest_education || '';
                        document.getElementById('employmentStatus').value = data.education_info.employment_status || '';
                        document.getElementById('occupation').value = data.education_info.occupation || '';
                    }
                    
                    if (data.voter_info) {
                        document.getElementById('isRegistered').checked = data.voter_info.is_registered === 1;
                        document.getElementById('precinctNumber').value = data.voter_info.precinct_number || '';
                    }
                    
                    document.querySelectorAll('.form-control.error').forEach(el => {
                        el.classList.remove('error');
                    });
                    document.querySelectorAll('.error-message.show').forEach(el => {
                        el.classList.remove('show');
                    });
                    
                    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
                    document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
                    document.querySelector('.tab[data-tab="personal"]').classList.add('active');
                    document.getElementById('personal-tab').classList.add('active');
                    
                    residentModal.style.display = 'block';
                } else {
                    showMessage('Error loading resident for editing: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error loading resident for editing: ' + error.message, 'error');
            }
        }
        
        async function saveResident(e) {
            e.preventDefault();
            
            if (!validateForm()) {
                showMessage('Please fix the errors in the form before saving.', 'error');
                return;
            }
            
            const residentId = document.getElementById('residentId').value;
            
            const formData = {
                first_name: document.getElementById('firstName').value.trim(),
                middle_name: document.getElementById('middleName').value.trim(),
                last_name: document.getElementById('lastName').value.trim(),
                birth_date: document.getElementById('birthDate').value,
                sex: document.getElementById('sex').value,
                civil_status: document.getElementById('civilStatus').value,
                contact_number: document.getElementById('contactNumber').value.trim(),
                house_no: document.getElementById('houseNo').value.trim(),
                street: document.getElementById('street').value.trim(),
                is_pwd: document.getElementById('isPwd').checked ? 1 : 0,
                is_senior: document.getElementById('isSenior').checked ? 1 : 0,
                is_solo_parent: document.getElementById('isSoloParent').checked ? 1 : 0,
                is_4ps: document.getElementById('is4ps').checked ? 1 : 0,
                
                health_info: {
                    blood_type: document.getElementById('bloodType').value,
                    known_allergies: document.getElementById('knownAllergies').value.trim(),
                    pre_existing_conditions: document.getElementById('preExistingConditions').value.trim(),
                    emergency_contact_name: document.getElementById('emergencyContactName').value.trim(),
                    emergency_contact_number: document.getElementById('emergencyContactNumber').value.trim(),
                    vaccination_status: document.getElementById('vaccinationStatus').value
                },
                
                education_info: {
                    highest_education: document.getElementById('highestEducation').value,
                    employment_status: document.getElementById('employmentStatus').value,
                    occupation: document.getElementById('occupation').value.trim()
                },
                
                voter_info: {
                    is_registered: document.getElementById('isRegistered').checked ? 1 : 0,
                    precinct_number: document.getElementById('precinctNumber').value.trim()
                }
            };
            
            try {
                let response;
                let endpoint;
                
                if (residentId) {
                    formData.resident_id = residentId;
                    endpoint = ENDPOINTS.UPDATE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'PUT',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(formData)
                    });
                } else {
                    endpoint = ENDPOINTS.SAVE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(formData)
                    });
                }
                
                const result = await response.json();
                
                if (result.success) {
                    showMessage(result.message, 'success');
                    loadResidents();
                    residentModal.style.display = 'none';
                } else {
                    showMessage('Error saving resident: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error saving resident: ' + error.message, 'error');
            }
        }
        
        async function deleteResident(id) {
            if (confirm('Are you sure you want to delete this resident from Barangay Immaculate records?\n\nThis action cannot be undone.')) {
                try {
                    const response = await fetch(ENDPOINTS.DELETE_RESIDENT, {
                        method: 'DELETE',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({ resident_id: id })
                    });
                    
                    const result = await response.json();
                    
                    if (result.success) {
                        showMessage(result.message, 'success');
                        loadResidents();
                    } else {
                        showMessage('Error deleting resident: ' + result.error, 'error');
                    }
                } catch (error) {
                    showMessage('Error deleting resident: ' + error.message, 'error');
                }
            }
        }
        
        function showMessage(message, type) {
            messageArea.innerHTML = `
                <div class="alert alert-${type}">
                    ${message}
                </div>
            `;
            
            setTimeout(() => {
                messageArea.innerHTML = '';
            }, 3000);
        }
        
        function formatDate(dateString) {
            if (!dateString) return 'N/A';
            
            const date = new Date(dateString);
            if (isNaN(date.getTime())) return 'N/A';
            
            const options = { year: 'numeric', month: 'long', day: 'numeric' };
            return date.toLocaleDateString(undefined, options);
        }
