        // Functions
        async function loadResidents() {
            try {
                const response = await fetch(ENDPOINTS.GET_RESIDENTS);
                const result = await response.json();
                
                if (result.success) {
                    displayResidents(result.data);
                    updateStats(result.data);
                } else {
                    showMessage('Error loading residents: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error loading residents: ' + error.message, 'error');
            }
        }
        
        function displayResidents(residents) {
            residentsTableBody.innerHTML = '';
            residentCount.textContent = residents.length;
            
            if (residents.length === 0) {
                residentsTableBody.innerHTML = `
                    <tr>

        function getResidentStatus(resident) {
            if (resident.is_pwd) return { text: 'PWD', class: 'pwd' };
            if (resident.is_senior) return { text: 'Senior', class: 'senior' };
            if (resident.is_solo_parent) return { text: 'Solo Parent', class: 'solo-parent' };
            if (resident.is_4ps) return { text: '4Ps', class: '4ps' };
            return { text: 'Regular', class: 'unknown' };
        }
        
        // FIXED: Handle both nested and flat data structures
        function getOccupation(resident) {
            // Check flat structure first (from get_residents.php with JOINs)
            if (resident.occupation && resident.occupation !== 'N/A' && resident.occupation !== '') {
                return resident.occupation;
            }
            
            // Check nested structure (from get_resident.php)
            if (resident.education_info && resident.education_info.occupation && 
                resident.education_info.occupation !== 'N/A' && resident.education_info.occupation !== '') {
                return resident.education_info.occupation;
            }
            
            // Check employment_status as fallback
            if (resident.employment_status && resident.employment_status !== 'N/A' && resident.employment_status !== '') {
                return resident.employment_status;
            }
            
            if (resident.education_info && resident.education_info.employment_status && 
                resident.education_info.employment_status !== 'N/A' && resident.education_info.employment_status !== '') {
                return resident.education_info.employment_status;
            }
            
            return 'N/A';
        }
        
        // FIXED: Handle both nested and flat data structures
        function getBloodType(resident) {
            // Check flat structure first (from get_residents.php with JOINs)
            if (resident.blood_type && resident.blood_type !== 'N/A' && resident.blood_type !== '') {
                return resident.blood_type;
            }
            
            // Check nested structure (from get_resident.php)
            if (resident.health_info && resident.health_info.blood_type && 
                resident.health_info.blood_type !== 'N/A' && resident.health_info.blood_type !== '') {
                return resident.health_info.blood_type;
            }
            
            return 'N/A';
        }
        
        function updateStats(residents) {
            // Update total residents
            totalResidentsEl.textContent = residents.length;
            
            // Count senior citizens
            const seniorCount = residents.filter(resident => {
                return calculateAge(resident.birth_date) >= 60 || resident.is_senior;
            }).length;
            seniorCitizensEl.textContent = seniorCount;
            
            // Count PWD residents

            pwdResidentsEl.textContent = pwdCount;

        function calculateAge(birthDate) {
            const today = new Date();
            const birth = new Date(birthDate);
            let age = today.getFullYear() - birth.getFullYear();
            const monthDiff = today.getMonth() - birth.getMonth();
            
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
                age--;
            }
            
            return age;
        }
        
        // UPDATED SEARCH FUNCTION - PREVENTS SCROLLBAR SHIFT
        function filterResidents() {
            const searchTerm = searchInput.value.toLowerCase();
            const rows = residentsTableBody.getElementsByTagName('tr');
            let visibleCount = 0;
            
            for (let i = 0; i < rows.length; i++) {
                const row = rows[i];
                const cells = row.getElementsByTagName('td');
                let found = false;
                
                // Skip loading/empty rows
                if (row.classList.contains('loading') || row.classList.contains('no-residents')) {
                    row.style.display = searchTerm === '' ? '' : 'none';
                    continue;
                }
                
                for (let j = 0; j < cells.length; j++) {
                    if (cells[j].textContent.toLowerCase().includes(searchTerm)) {
                        found = true;
                        break;
                    }
                }
                
                if (found) {
                    row.style.display = '';
                    row.classList.remove('hidden-row');
                    visibleCount++;
                } else {
                    // Instead of display:none which causes shift, use opacity and height
                    row.style.opacity = '0.3';
                    row.style.height = '70px'; // Maintain the same height
                    row.style.overflow = 'hidden';
                    row.classList.add('hidden-row');
                }
            }
            
            // Update visible count if needed
            if (searchTerm === '') {
                residentCount.textContent = rows.length;
            } else {
                residentCount.textContent = visibleCount;
            }
        }

        async function saveResident(e) {
            e.preventDefault();
            
            const residentId = document.getElementById('residentId').value;
            
            // Collect all form data including health, education, and voter info
            const formData = {
                first_name: document.getElementById('firstName').value,
                middle_name: document.getElementById('middleName').value,
                last_name: document.getElementById('lastName').value,
                birth_date: document.getElementById('birthDate').value,
                sex: document.getElementById('sex').value,
                civil_status: document.getElementById('civilStatus').value,
                contact_number: document.getElementById('contactNumber').value,
                house_no: document.getElementById('houseNo').value,
                street: document.getElementById('street').value,
                is_pwd: document.getElementById('isPwd').checked ? 1 : 0,
                is_senior: document.getElementById('isSenior').checked ? 1 : 0,
                is_solo_parent: document.getElementById('isSoloParent').checked ? 1 : 0,
                is_4ps: document.getElementById('is4ps').checked ? 1 : 0,
                
                // Health information
                health_info: {
                    blood_type: document.getElementById('bloodType').value,
                    known_allergies: document.getElementById('knownAllergies').value,
                    pre_existing_conditions: document.getElementById('preExistingConditions').value,
                    emergency_contact_name: document.getElementById('emergencyContactName').value,
                    emergency_contact_number: document.getElementById('emergencyContactNumber').value,
                    vaccination_status: document.getElementById('vaccinationStatus').value
                },
                
                // Education & employment information
                education_info: {
                    highest_education: document.getElementById('highestEducation').value,
                    employment_status: document.getElementById('employmentStatus').value,
                    occupation: document.getElementById('occupation').value
                },
                
                // Voter information
                voter_info: {
                    is_registered: document.getElementById('isRegistered').checked ? 1 : 0,
                    precinct_number: document.getElementById('precinctNumber').value
                }
            };
            
            try {
                let response;
                let endpoint;
                
                if (residentId) {
                    // Update existing resident
                    formData.resident_id = residentId;
                    endpoint = ENDPOINTS.UPDATE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'PUT',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(formData)
                    });
                } else {
                    // Add new resident
                    endpoint = ENDPOINTS.SAVE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify(formData)
                    });
                }
                
                const result = await response.json();
                
                if (result.success) {
                    showMessage(result.message, 'success');
                    loadResidents();
                    residentModal.style.display = 'none';
                } else {
                    showMessage('Error saving resident: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error saving resident: ' + error.message, 'error');
            }
        }

                    }

                }
            }
        }

        function formatDate(dateString) {
            const options = { year: 'numeric', month: 'long', day: 'numeric' };
            return new Date(dateString).toLocaleDateString(undefined, options);
        }
