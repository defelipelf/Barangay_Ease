        function setupFormValidation() {
            // Mobile number - only numbers

            const emergencyMobileInput = document.getElementById('emergencyContactNumber');
            const precinctInput = document.getElementById('precinctNumber');
            const birthDateInput = document.getElementById('birthDate');

            // Set min date (assuming no one older than 120 years)

                // Validate length (minimum 7, maximum 15 digits for phone numbers)
                const digitsOnly = value.replace(/\D/g, '');
                if (digitsOnly.length > 11) {
                    value = value.substring(0, value.length - 1);
                }
                
                e.target.value = value;
                validateMobileNumber(e.target);
            });
            
            // Emergency contact number validation
            emergencyMobileInput.addEventListener('input', function(e) {
                let value = e.target.value;
                // Allow only numbers, spaces, plus, hyphen, and parentheses
                value = value.replace(/[^\d\s+\-()]/g, '');
                
                // Validate length
                const digitsOnly = value.replace(/\D/g, '');
                if (digitsOnly.length > 11) {
                    value = value.substring(0, value.length - 1);

                e.target.value = value;

        // Validate mobile number format
        function validateMobileNumber(input) {
            const value = input.value;
            const digitsOnly = value.replace(/\D/g, '');
            const errorMessage = input.parentElement.querySelector('.error-message');
            
            if (value && digitsOnly.length < 7) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number must be at least 7 digits';
                    errorMessage.classList.add('show');
                }
                return false;
            } else if (value && digitsOnly.length > 15) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number cannot exceed 15 digits';
                    errorMessage.classList.add('show');
                }
                return false;
            } else if (value && !/^[\d\s+\-()]+$/.test(value)) {
                input.classList.add('error');
                if (errorMessage) {
                    errorMessage.textContent = 'Mobile number can only contain numbers, spaces, +, -, and ()';
                    errorMessage.classList.add('show');
                }
                return false;
            } else {
                input.classList.remove('error');
                if (errorMessage) {
                    errorMessage.classList.remove('show');
                }
                return true;
            }
        }
        
        // Validate entire form
        function validateForm() {
            let isValid = true;
            
            // Clear previous error states
            document.querySelectorAll('.form-control.error').forEach(el => {
                el.classList.remove('error');
            });
            document.querySelectorAll('.error-message.show').forEach(el => {
                el.classList.remove('show');
            });
            
            // Validate required fields
            const requiredFields = ['firstName', 'lastName', 'birthDate', 'sex', 'civilStatus', 'houseNo', 'street'];
            requiredFields.forEach(fieldId => {
                const field = document.getElementById(fieldId);
                const errorMessage = field.parentElement.querySelector('.error-message');
                if (field && !field.value.trim()) {
                    field.classList.add('error');
                    if (errorMessage) {
                        errorMessage.textContent = 'This field is required';
                        errorMessage.classList.add('show');
                    }
                    isValid = false;
                }
            });
            
            // Validate mobile number if provided
            const mobileInput = document.getElementById('contactNumber');
            if (mobileInput.value.trim()) {
                if (!validateMobileNumber(mobileInput)) {
                    isValid = false;
                }
            }
            
            // Validate emergency contact number if provided
            const emergencyInput = document.getElementById('emergencyContactNumber');
            if (emergencyInput.value.trim()) {
                const digitsOnly = emergencyInput.value.replace(/\D/g, '');
                if (digitsOnly.length > 0 && digitsOnly.length < 7) {
                    emergencyInput.classList.add('error');
                    const errorMessage = emergencyInput.parentElement.querySelector('.error-message');
                    if (errorMessage) {
                        errorMessage.textContent = 'Emergency contact number must be at least 7 digits';
                        errorMessage.classList.add('show');
                    }
                    isValid = false;
                }
            }
            
            // Validate birth date is not in future
            const birthDate = new Date(document.getElementById('birthDate').value);
            if (birthDate > new Date()) {
                const birthDateInput = document.getElementById('birthDate');
                birthDateInput.classList.add('error');
                const errorMessage = birthDateInput.parentElement.querySelector('.error-message');
                if (errorMessage) {
                    errorMessage.textContent = 'Birth date cannot be in the future';
                    errorMessage.classList.add('show');
                }
                isValid = false;
            }
            
            // Validate birth date is not too far in the past (max 120 years)
            const minDate = new Date();
            minDate.setFullYear(minDate.getFullYear() - 120);
            if (birthDate < minDate) {
                const birthDateInput = document.getElementById('birthDate');
                birthDateInput.classList.add('error');
                const errorMessage = birthDateInput.parentElement.querySelector('.error-message');
                if (errorMessage) {
                    errorMessage.textContent = 'Birth date cannot be more than 120 years ago';
                    errorMessage.classList.add('show');
                }
                isValid = false;
            }
            
            return isValid;
        }

        // Display residents with pagination
        function displayResidents() {
            residentsTableBody.innerHTML = '';
            
            if (!filteredResidents || filteredResidents.length === 0) {
                residentsTableBody.innerHTML = `
                    <tr>
                        <td colspan="6" class="no-residents">
                            <h3>No Residents Found</h3>
                            <p>Click "Add Resident" to register a new resident.</p>
                        </td>
                    </tr>
                `;
                residentCount.textContent = '0';
                totalResidentsEl.textContent = '0';
                updatePagination(0);
                return;
            }
            
            residentCount.textContent = filteredResidents.length;
            totalResidentsEl.textContent = filteredResidents.length;
            
            // Pagination logic
            const startIndex = (currentPage - 1) * rowsPerPage;
            const endIndex = Math.min(startIndex + rowsPerPage, filteredResidents.length);
            const paginatedResidents = filteredResidents.slice(startIndex, endIndex);
            
            paginatedResidents.forEach(resident => {
                const age = calculateAge(resident.birth_date);
                
                // Get status badges
                let statusHtml = '';
                if (resident.is_senior) statusHtml += '<span class="status-badge status-senior">Senior</span> ';
                if (resident.is_pwd) statusHtml += '<span class="status-badge status-pwd">PWD</span> ';
                if (resident.is_4ps) statusHtml += '<span class="status-badge status-4ps">4Ps</span> ';
                if (resident.is_solo_parent) statusHtml += '<span class="status-badge status-solo">Solo Parent</span> ';
                if (!statusHtml) statusHtml = '<span class="status-badge status-regular">Regular</span>';
                
                // Format contact number
                let contactNumber = resident.contact_number || 'N/A';
                if (contactNumber.length > 15) {
                    contactNumber = contactNumber.substring(0, 12) + '...';
                }
                
                const row = document.createElement('tr');
                row.innerHTML = `
                    <td class="name-cell">${resident.last_name}, ${resident.first_name} ${resident.middle_name ? resident.middle_name.charAt(0) + '.' : ''}</td>
                    <td>${age}</td>
                    <td class="address-cell tooltip" title="${resident.house_no} ${resident.street}">

                        <button class="btn btn-danger btn-sm" onclick="deleteResident(${resident.resident_id})">Del</button>
                    </td>
                `;
                residentsTableBody.appendChild(row);
            });
            
            updatePagination(filteredResidents.length);
        }
        
        // Update pagination controls
        function updatePagination(totalItems) {
            const totalPages = Math.ceil(totalItems / rowsPerPage) || 1;
            
            showingStart.textContent = totalItems ? (currentPage - 1) * rowsPerPage + 1 : 0;
            showingEnd.textContent = totalItems ? Math.min(currentPage * rowsPerPage, totalItems) : 0;
            
            prevPageBtn.disabled = currentPage === 1;
            nextPageBtn.disabled = currentPage === totalPages || totalItems === 0;
            
            // Generate page numbers
            paginationNumbers.innerHTML = '';
            for (let i = 1; i <= totalPages && i <= 10; i++) {
                const pageSpan = document.createElement('span');
                pageSpan.className = `page-number ${i === currentPage ? 'active' : ''}`;
                pageSpan.textContent = i;
                pageSpan.addEventListener('click', () => {
                    currentPage = i;
                    displayResidents();
                });
                paginationNumbers.appendChild(pageSpan);
            }
            
            if (totalPages > 10) {
                const ellipsis = document.createElement('span');
                ellipsis.className = 'page-number';
                ellipsis.textContent = '...';
                paginationNumbers.appendChild(ellipsis);
                
                const lastPage = document.createElement('span');
                lastPage.className = `page-number ${totalPages === currentPage ? 'active' : ''}`;
                lastPage.textContent = totalPages;
                lastPage.addEventListener('click', () => {
                    currentPage = totalPages;
                    displayResidents();
                });
                paginationNumbers.appendChild(lastPage);
            }
        }
        
        // Pagination functions

            if (currentPage > 1) {

        function goToNextPage() {
            const totalPages = Math.ceil(filteredResidents.length / rowsPerPage);
            if (currentPage < totalPages) {

            }
        }

        // Filter by category
        function filterByCategory(category) {
            currentPage = 1;
            
            if (category === 'all') {
                filteredResidents = [...allResidents];
            } else {
                filteredResidents = allResidents.filter(resident => {
                    switch(category) {
                        case 'senior': return resident.is_senior;
                        case 'pwd': return resident.is_pwd;
                        case '4ps': return resident.is_4ps;
                        case 'solo': return resident.is_solo_parent;
                        default: return true;
                    }
                });
            }
            
            displayResidents();
        }
        
        // Search filter
        function filterResidents() {
            const searchTerm = searchInput.value.toLowerCase();
            
            if (!searchTerm) {
                filteredResidents = [...allResidents];
            } else {
                filteredResidents = allResidents.filter(resident => {
                    const fullName = `${resident.first_name} ${resident.middle_name || ''} ${resident.last_name}`.toLowerCase();
                    const address = `${resident.house_no} ${resident.street}`.toLowerCase();
                    const contact = (resident.contact_number || '').toLowerCase();
                    
                    return fullName.includes(searchTerm) || 
                           address.includes(searchTerm) || 
                           contact.includes(searchTerm);
                });
            }
            
            currentPage = 1;
            displayResidents();
            
            // Reset active filter pill
            document.querySelectorAll('.filter-pill').forEach(p => p.classList.remove('active'));
            document.querySelector('[data-filter="all"]').classList.add('active');
            currentFilter = 'all';
        }
        
        // Calculate age
        function calculateAge(birthDate) {
            if (!birthDate) return 'N/A';
            
            const today = new Date();
            const birth = new Date(birthDate);
            
            if (isNaN(birth.getTime())) return 'N/A';
            
            let age = today.getFullYear() - birth.getFullYear();
            const monthDiff = today.getMonth() - birth.getMonth();
            
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
                age--;
            }
            
            return age >= 0 ? age : 'N/A';
        }

        // Save resident
        async function saveResident(e) {
            e.preventDefault();
            
            if (!validateForm()) {
                showMessage('Please fix the errors in the form before saving.', 'error');
                return;
            }
            
            const residentId = document.getElementById('residentId').value;
            
            const formData = {
                first_name: document.getElementById('firstName').value.trim(),
                middle_name: document.getElementById('middleName').value.trim(),
                last_name: document.getElementById('lastName').value.trim(),
                birth_date: document.getElementById('birthDate').value,
                sex: document.getElementById('sex').value,
                civil_status: document.getElementById('civilStatus').value,
                contact_number: document.getElementById('contactNumber').value.trim(),
                house_no: document.getElementById('houseNo').value.trim(),
                street: document.getElementById('street').value.trim(),
                is_pwd: document.getElementById('isPwd').checked ? 1 : 0,
                is_senior: document.getElementById('isSenior').checked ? 1 : 0,
                is_solo_parent: document.getElementById('isSoloParent').checked ? 1 : 0,
                is_4ps: document.getElementById('is4ps').checked ? 1 : 0,
                
                health_info: {
                    blood_type: document.getElementById('bloodType').value,
                    known_allergies: document.getElementById('knownAllergies').value.trim(),
                    pre_existing_conditions: document.getElementById('preExistingConditions').value.trim(),
                    emergency_contact_name: document.getElementById('emergencyContactName').value.trim(),
                    emergency_contact_number: document.getElementById('emergencyContactNumber').value.trim(),
                    vaccination_status: document.getElementById('vaccinationStatus').value
                },
                
                education_info: {
                    highest_education: document.getElementById('highestEducation').value,
                    employment_status: document.getElementById('employmentStatus').value,
                    occupation: document.getElementById('occupation').value.trim()
                },
                
                voter_info: {
                    is_registered: document.getElementById('isRegistered').checked ? 1 : 0,
                    precinct_number: document.getElementById('precinctNumber').value.trim()
                }
            };
            
            try {
                let response;
                let endpoint;
                
                if (residentId) {
                    formData.resident_id = residentId;
                    endpoint = ENDPOINTS.UPDATE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'PUT',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(formData)
                    });
                } else {
                    endpoint = ENDPOINTS.SAVE_RESIDENT;
                    response = await fetch(endpoint, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(formData)
                    });
                }
                
                const result = await response.json();
                
                if (result.success) {
                    showMessage(result.message, 'success');
                    loadResidents();
                    residentModal.style.display = 'none';
                } else {
                    showMessage('Error saving resident: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error saving resident: ' + error.message, 'error');
            }
        }

        // Format date
        function formatDate(dateString) {
            if (!dateString) return 'N/A';
            const date = new Date(dateString);
            if (isNaN(date.getTime())) return 'N/A';
            return date.toLocaleDateString(undefined, { year: 'numeric', month: 'long', day: 'numeric' });
