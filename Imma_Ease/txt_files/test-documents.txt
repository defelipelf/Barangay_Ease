        // Initialize the application
        document.addEventListener('DOMContentLoaded', function() {
            checkAuthentication();
            loadResidents();
            loadTemplates();
            setupEventListeners();
            
            // Set first document type as active by default
            document.querySelector('.document-type').classList.add('active');
        });

        function checkAuthentication() {
            const user = JSON.parse(localStorage.getItem('user'));
            if (!user) {
                window.location.href = 'login.html';
                return;
            }
        }

        function setupEventListeners() {
            // Document type selection
            document.querySelectorAll('.document-type').forEach(card => {
                card.addEventListener('click', function() {
                    document.querySelectorAll('.document-type').forEach(c => c.classList.remove('active'));
                    this.classList.add('active');
                    selectedDocumentType = this.getAttribute('data-document-type');
                    updateGenerateButton();
                    showAdditionalFields();
                });
            });

            // Search functionality
            document.getElementById('residentSearch').addEventListener('input', function() {
                const searchTerm = this.value.toLowerCase().trim();
                
                if (searchTerm.length < 2) {
                    document.getElementById('searchResults').style.display = 'none';
                    return;
                }

                const filteredResidents = residents.filter(resident => {
                    const fullName = `${resident.first_name} ${resident.middle_name || ''} ${resident.last_name}`.toLowerCase();
                    return fullName.includes(searchTerm) || 
                           resident.last_name.toLowerCase().includes(searchTerm) ||
                           resident.first_name.toLowerCase().includes(searchTerm);
                });

                displaySearchResults(filteredResidents);
            });

            // Generate document button
            document.getElementById('generateDocumentBtn').addEventListener('click', generateDocument);

            // Print and download buttons
            document.getElementById('printBtn').addEventListener('click', printDocument);
            document.getElementById('downloadPdfBtn').addEventListener('click', downloadPDF);
            document.getElementById('newDocumentBtn').addEventListener('click', newDocument);

            // Hide search results when clicking outside
            document.addEventListener('click', function(e) {
                if (!e.target.closest('.search-container')) {
                    document.getElementById('searchResults').style.display = 'none';
                }
            });
        }

        async function loadResidents() {
            try {
                const response = await fetch(ENDPOINTS.RESIDENTS);
                const result = await response.json();
                
                if (result.success) {
                    residents = result.data;
                    document.getElementById('totalResidents').textContent = residents.length;
                } else {
                    showMessage('Error loading residents: ' + result.error, 'error');
                }
            } catch (error) {
                showMessage('Error loading residents: ' + error.message, 'error');
            }
        }

        async function loadTemplates() {
            try {
                const response = await fetch(ENDPOINTS.DOCUMENT_TEMPLATES);
                const result = await response.json();
                
                if (result.success) {
                    templates = result.templates;
                } else {
                    templates = fallbackTemplates;
                }
            } catch (error) {
                templates = fallbackTemplates;
            }
        }

        function displaySearchResults(filteredResidents) {
            const searchResults = document.getElementById('searchResults');
            
            if (filteredResidents.length === 0) {
                searchResults.innerHTML = '<div class="search-result-item">No residents found</div>';
            } else {
                searchResults.innerHTML = filteredResidents.map(resident => `
                    <div class="search-result-item" data-resident-id="${resident.resident_id}">
                        <strong>${resident.last_name}, ${resident.first_name}</strong>
                        ${resident.middle_name ? resident.middle_name + ' ' : ''}<br>
                        <small>${resident.contact_number || 'No contact number'} • ${resident.house_no} ${resident.street}</small>
                    </div>
                `).join('');

                searchResults.querySelectorAll('.search-result-item').forEach(item => {
                    item.addEventListener('click', function() {
                        const residentId = this.getAttribute('data-resident-id');
                        const resident = residents.find(r => r.resident_id == residentId);
                        if (resident) {
                            selectResident(resident);
                        }
                    });
                });
            }
            
            searchResults.style.display = 'block';
        }

        function selectResident(resident) {
            selectedResident = resident;
            document.getElementById('residentSearch').value = '';
            document.getElementById('searchResults').style.display = 'none';
            
            const selectedResidentDiv = document.getElementById('selectedResident');
            selectedResidentDiv.innerHTML = `
                <div class="resident-info">
                    <div class="resident-avatar">
                        <i class="fas fa-user"></i>
                    </div>
                    <div class="resident-details">
                        <h4>${resident.last_name}, ${resident.first_name} ${resident.middle_name || ''}</h4>
                        <p><i class="fas fa-map-pin" style="font-size: 11px;"></i> ${resident.house_no} ${resident.street} • ${resident.contact_number || 'No contact'}</p>
                    </div>
                </div>
                <button class="change-btn" onclick="clearSelectedResident()">
                    <i class="fas fa-times"></i> Change
                </button>
            `;
            selectedResidentDiv.style.display = 'flex';
            
            updateGenerateButton();
        }

        window.clearSelectedResident = function() {
            selectedResident = null;
            document.getElementById('selectedResident').style.display = 'none';
            document.getElementById('residentSearch').value = '';
            updateGenerateButton();
        };

        function updateGenerateButton() {
            const generateBtn = document.getElementById('generateDocumentBtn');
            generateBtn.disabled = !(selectedDocumentType && selectedResident);
        }

        function showAdditionalFields() {
            const additionalFields = document.getElementById('additionalFields');
            const yearsResidingField = document.getElementById('yearsResidingField');
            
            document.getElementById('purpose').value = '';
            document.getElementById('yearsResiding').value = '';
            
            if (selectedDocumentType === 'residency_certificate') {
                yearsResidingField.style.display = 'block';
            } else {
                yearsResidingField.style.display = 'none';
            }
            
            additionalFields.style.display = 'block';
        }

        async function generateDocument() {
            if (!selectedDocumentType || !selectedResident) {
                showMessage('Please select both a document type and a resident', 'error');
                return;
            }

            try {
                const response = await fetch(`${ENDPOINTS.RESIDENT_DATA}?resident_id=${selectedResident.resident_id}`);
                const result = await response.json();
                
                if (!result.success) {
                    throw new Error(result.error);
                }

                const residentData = result.data;
                
                const purpose = document.getElementById('purpose').value;
                const yearsResiding = document.getElementById('yearsResiding').value;

                if (!purpose && selectedDocumentType !== 'residency_certificate') {
                    showMessage('Please enter the purpose of the document', 'error');
                    return;
                }

                if (selectedDocumentType === 'residency_certificate' && !yearsResiding) {
                    showMessage('Please enter the number of years residing in the barangay', 'error');
                    return;
                }

                const currentDate = new Date().toLocaleDateString('en-US', {
                    year: 'numeric',
                    month: 'long',
                    day: 'numeric'
                });

                currentDocumentData = {
                    ...residentData,
                    purpose: purpose,
                    years_residing: yearsResiding,
                    date_issued: currentDate
                };

                const template = templates[selectedDocumentType];
                if (!template) {
                    throw new Error('Template not found for selected document type');
                }

                let documentHtml = template.template;
                
                for (const [key, value] of Object.entries(currentDocumentData)) {
                    const placeholder = `{{${key}}}`;
                    documentHtml = documentHtml.replace(new RegExp(placeholder, 'g'), value || '');
                }

                document.getElementById('previewContent').innerHTML = documentHtml;
                document.getElementById('documentPreview').style.display = 'block';
                
                document.getElementById('documentPreview').scrollIntoView({ behavior: 'smooth' });
                
                showMessage('Document generated successfully!', 'success');
                
            } catch (error) {
                showMessage('Error generating document: ' + error.message, 'error');
                console.error('Document generation error:', error);
            }
        }

        function printDocument() {
            const content = document.getElementById('previewContent').innerHTML;
            const printWindow = window.open('', '_blank');
            printWindow.document.write(`
                <!DOCTYPE html>
                <html>
                <head>
                    <title>Print Document</title>
                    <style>
                        body { 
                            font-family: 'Times New Roman', Times, serif; 
                            margin: 0; 
                            padding: 20mm;
                            line-height: 1.8;
                        }
                        .document-template { 
                            width: 100%; 
                            min-height: 100vh;
                        }
                        @media print {
                            body { margin: 0; padding: 20mm; }
                        }
                    </style>
                </head>
                <body>
                    ${content}
                    <script>
                        window.onload = function() {
                            window.print();
                            setTimeout(() => window.close(), 500);
                        }
                    <\/script>
                </body>
                </html>
            `);
            printWindow.document.close();
        }

        async function downloadPDF() {
            const loadingElement = document.getElementById('pdfLoading');
            const downloadBtn = document.getElementById('downloadPdfBtn');
            
            try {
                loadingElement.style.display = 'block';
                downloadBtn.disabled = true;
                downloadBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Generating...';

                const element = document.getElementById('previewContent');
                
                const canvas = await html2canvas(element, {
                    scale: 2,
                    useCORS: true,
                    logging: false,
                    backgroundColor: '#ffffff'
                });

                const imgData = canvas.toDataURL('image/jpeg', 1.0);
                
                const { jsPDF } = window.jspdf;
                const pdf = new jsPDF('p', 'mm', 'a4');
                
                const imgWidth = 210;
                const imgHeight = (canvas.height * imgWidth) / canvas.width;
                
                pdf.addImage(imgData, 'JPEG', 0, 0, imgWidth, imgHeight);

                const docName = templates[selectedDocumentType].name.toLowerCase().replace(/\s+/g, '_');
                const residentName = selectedResident.last_name.toLowerCase();
                const filename = `${docName}_${residentName}_${new Date().getTime()}.pdf`;

                pdf.save(filename);
                showMessage('PDF downloaded successfully!', 'success');
                
            } catch (error) {
                showMessage('Error generating PDF: ' + error.message, 'error');
                console.error('PDF generation error:', error);
            } finally {
                loadingElement.style.display = 'none';
                downloadBtn.disabled = false;
                downloadBtn.innerHTML = '<i class="fas fa-download"></i> PDF';
            }
        }

        function newDocument() {
            selectedDocumentType = 'indigency_certificate';
            selectedResident = null;
            currentDocumentData = {};
            
            document.querySelectorAll('.document-type').forEach(card => {
                card.classList.remove('active');
                if (card.getAttribute('data-document-type') === 'indigency_certificate') {
                    card.classList.add('active');
                }
            });
            
            document.getElementById('selectedResident').style.display = 'none';
            document.getElementById('residentSearch').value = '';
            document.getElementById('additionalFields').style.display = 'none';
            document.getElementById('documentPreview').style.display = 'none';
            document.getElementById('purpose').value = '';
            document.getElementById('yearsResiding').value = '';
            
            updateGenerateButton();
            showMessage('Ready to create new document', 'success');
        }

        function showMessage(message, type) {
            const messageArea = document.getElementById('messageArea');
            messageArea.innerHTML = `
                <div class="alert alert-${type}">
                    <i class="fas fa-${type === 'success' ? 'check-circle' : 'exclamation-circle'}"></i>
                    ${message}
                </div>
            `;
            
            setTimeout(() => {
                messageArea.innerHTML = '';
            }, 5000);
        }
